🎯 GOAL: Switch from Zarela's private IPFS to free public IPFS

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

📋 OPTION 1: Use the Update Script (Easiest)

   ./scripts/update-ipfs-config.sh

   Then choose option 1 (Infura + Cloudflare - Recommended)

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

📝 OPTION 2: Manual Update

   Edit your .env file and replace:

   FROM:
   REACT_APP_IPFS=https://ipfs.zarela.io/
   REACT_APP_IPFS_GET_LINK=https://get-ipfs.zarela.io/ipfs/

   TO:
   REACT_APP_IPFS=https://ipfs.infura.io:5001
   REACT_APP_IPFS_GET_LINK=https://cloudflare-ipfs.com/ipfs/

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
